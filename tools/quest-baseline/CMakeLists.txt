cmake_minimum_required(VERSION 3.22)
project(PawsQuestBaseline LANGUAGES CXX)
include(CTest)
set(PAWS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(WEBENGINE_BUILD_MODULES ON)
set(WEBENGINE_BUILD_QUESTS ON)
add_subdirectory("${PAWS_ROOT}/external/WebEngine/modules" engine-modules)
add_subdirectory("${PAWS_ROOT}/external/WebEngine/quests" engine-quests)
add_library(paws_baseline_rules STATIC
  ${PAWS_ROOT}/src/puzzle.cpp ${PAWS_ROOT}/src/content.cpp ${PAWS_ROOT}/src/afternoon.cpp
  ${PAWS_ROOT}/src/release1/rules/puzzles.cpp ${PAWS_ROOT}/src/release1/rules/content.cpp
  ${PAWS_ROOT}/src/release1/rules/beach_content.cpp
  ${PAWS_ROOT}/src/release1/progression/policy.cpp ${PAWS_ROOT}/src/release1/progression/codec.cpp
  ${PAWS_ROOT}/src/release1/progression/quest_adapter.cpp)
target_link_libraries(paws_baseline_rules PUBLIC WebEngine::Quests)
target_compile_features(paws_baseline_rules PUBLIC cxx_std_20)
target_include_directories(paws_baseline_rules PUBLIC ${PAWS_ROOT}/src)
target_compile_definitions(paws_baseline_rules PUBLIC PAWS_RELEASE1_PROGRESSION_PRODUCTION=1)
add_executable(paws_quest_baseline baseline.cpp)
target_link_libraries(paws_quest_baseline PRIVATE paws_baseline_rules)
target_compile_definitions(paws_quest_baseline PRIVATE PAWS_ROOT="${PAWS_ROOT}")
add_executable(paws_baseline_progression ${PAWS_ROOT}/tests/release1/progression/tests.cpp)
add_executable(paws_quest_pilots ${PAWS_ROOT}/tests/quest_pilots_tests.cpp)
add_executable(paws_quest_catalogue ${PAWS_ROOT}/tests/quest_catalogue_tests.cpp)
target_link_libraries(paws_quest_catalogue PRIVATE paws_baseline_rules)
target_compile_definitions(paws_quest_catalogue PRIVATE PAWS_ROOT="${PAWS_ROOT}")
target_link_libraries(paws_quest_pilots PRIVATE paws_baseline_rules)
target_compile_definitions(paws_quest_pilots PRIVATE PAWS_ROOT="${PAWS_ROOT}")
if(EMSCRIPTEN)
  foreach(target engine_modules engine_quests paws_baseline_rules paws_quest_pilots paws_quest_catalogue paws_quest_baseline paws_baseline_progression)
    target_compile_options(${target} PRIVATE -fexceptions)
  endforeach()
  foreach(target paws_quest_pilots paws_quest_catalogue paws_quest_baseline paws_baseline_progression)
    set_target_properties(${target} PROPERTIES SUFFIX ".js")
    target_link_options(${target} PRIVATE -fexceptions -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=1048576 -sNODERAWFS=1)
  endforeach()
endif()
target_link_libraries(paws_baseline_progression PRIVATE paws_baseline_rules)
add_test(NAME quest_baseline COMMAND paws_quest_baseline --verify "${PAWS_ROOT}/tests/fixtures/quest-migration-m0")
add_test(NAME quest_current_baseline COMMAND paws_quest_baseline --verify-current "${PAWS_ROOT}/tests/fixtures/quest-migration-m5")
add_test(NAME baseline_progression COMMAND paws_baseline_progression)
add_test(NAME quest_pilots COMMAND paws_quest_pilots)
add_test(NAME quest_catalogue COMMAND paws_quest_catalogue)
